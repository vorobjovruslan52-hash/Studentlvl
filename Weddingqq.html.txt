        // Feature 1: Dynamic Status & Timer Tracker
        function updateLiveTracker() {
            const now = new Date();
            const currentHours = now.getHours();
            const currentMinutes = now.getMinutes();
            const nowSec = currentHours * 3600 + currentMinutes * 60 + now.getSeconds();


            const pairs = [
                { num: 1, start: 8*3600 + 30*60, end: 9*3600 + 50*60 },
                { num: 2, start: 10*3600, end: 11*3600 + 20*60 },
                { num: 3, start: 11*3600 + 30*60, end: 12*3600 + 50*60 },
                { num: 4, start: 13*3600, end: 14*3600 + 20*60 }
            ];


            let activePair = null;
            let nextBreak = null;


            for (let i = 0; i < pairs.length; i++) {
                if (nowSec >= pairs[i].start && nowSec <= pairs[i].end) {
                    activePair = pairs[i];
                    break;
                }
                if (i < pairs.length - 1 && nowSec > pairs[i].end && nowSec < pairs[i+1].start) {
                    nextBreak = { end: pairs[i+1].start, nextNum: pairs[i+1].num };
                    break;
                }
            }


            const titleEl = document.getElementById('current-status-title');
            const descEl = document.getElementById('current-status-desc');
            const timerEl = document.getElementById('timer-badge');


            if (activePair) {
                const diff = activePair.end - nowSec;
                const m = Math.floor(diff / 60);
                const s = diff % 60;
                titleEl.innerText = `Зараз іде ${activePair.num}-ша пара`;
                descEl.innerText = `До кінця пари залишилося:`;
                timerEl.innerText = `${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`;
            } else if (nextBreak) {
                const diff = nextBreak.end - nowSec;
                const m = Math.floor(diff / 60);
                const s = diff % 60;
                titleEl.innerText = `Перерва перед ${nextBreak.nextNum}-ю парою`;
                descEl.innerText = `До початку пари залишилося:`;
                timerEl.innerText = `${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`;
            } else {
                titleEl.innerText = `Заняття завершені або ще не почалися`;
                descEl.innerText = `Гарного відпочинку!`;
                timerEl.innerText = `--:--`;
            }
        }

        function renderTodos() {
            const list = document.getElementById('todo-list');
            list.innerHTML = '';
            todos.forEach((todo, idx) => {
                const li = document.createElement('li');
                li.className = `todo-item ${todo.completed ? 'completed' : ''}`;
                li.innerHTML = `
                    <div class="todo-left">
                        <input type="checkbox" ${todo.completed ? 'checked' : ''} onchange="toggleTodo(${idx})">
                        <span>${todo.text}</span>

                    <button class="btn-delete" onclick="deleteTodo(${idx})"><i class="fa-solid fa-trash"></i></button>
                `;
                list.appendChild(li);
            });
            localStorage.setItem('student_todos', JSON.stringify(todos));

        function addTodo() {
            const input = document.getElementById('todo-input');
            if (input.value.trim() !== '') {
                todos.push({ text: input.value.trim(), completed: false });
                input.value = '';
                renderTodos();
            }
        }


        function toggleTodo(index) {
            todos[index].completed = !todos[index].completed;
            renderTodos();
        }


        function deleteTodo(index) {
            todos.splice(index, 1);
            renderTodos();
        }

        function renderGrades() {
            const list = document.getElementById('grade-list');
            list.innerHTML = '';
            let total = 0;


            grades.forEach((g, i) => {
                total += Number(g.grade);
                const div = document.createElement('div');
                div.className = 'grade-item';
                div.innerHTML = `
                    <span>${g.subject}</span>
                    <strong>${g.grade} <i class="fa-solid fa-xmark" style="color:var(--danger-color); cursor:pointer; margin-left:5px;" onclick="deleteGrade(${i})"></i></strong>
                `;
                list.appendChild(div);
            });


            const gpa = grades.length > 0 ? (total / grades.length).toFixed(2) : '0.00';
            document.getElementById('gpa-val').innerText = gpa;
            localStorage.setItem('student_grades', JSON.stringify(grades));
        }


        function addGrade() {
            const sub = document.getElementById('subject-name').value.trim();
            const gr = document.getElementById('subject-grade').value;
            if (sub && gr) {
                grades.push({ subject: sub, grade: Number(gr) });
                document.getElementById('subject-name').value = '';
                document.getElementById('subject-grade').value = '';
                renderGrades();
            }
        }


        function deleteGrade(idx) {

        }

        // Theme Toggle

            }
        }

            } else {
